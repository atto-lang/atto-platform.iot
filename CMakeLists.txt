cmake_minimum_required(VERSION 3.20)
project(ores_iot_frame VERSION 0.1.0 LANGUAGES C)
include(CTest)

add_library(ores_iot_frame STATIC src/ores_iot_frame.c src/ores_iot_stream.c src/ores_iot_replay.c)
target_include_directories(ores_iot_frame PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(ores_iot_frame PUBLIC c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ores_iot_frame PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(test_iot_frame tests/test_iot_frame.c)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_iot_frame PRIVATE -UNDEBUG)
  endif()
  target_link_libraries(test_iot_frame PRIVATE ores_iot_frame)
  add_test(NAME iot_frame_contract COMMAND test_iot_frame)
  add_executable(test_iot_stream tests/test_iot_stream.c)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_iot_stream PRIVATE -UNDEBUG)
  endif()
  target_link_libraries(test_iot_stream PRIVATE ores_iot_frame)
  add_test(NAME iot_stream_contract COMMAND test_iot_stream)
  add_executable(test_iot_replay tests/test_iot_replay.c)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_iot_replay PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
  endif()
  target_link_libraries(test_iot_replay PRIVATE ores_iot_frame)
  add_test(NAME iot_replay_differential_40000 COMMAND test_iot_replay)
  add_executable(iot_host_demo examples/host_demo.c)
  target_link_libraries(iot_host_demo PRIVATE ores_iot_frame)
  add_test(NAME iot_host_demo COMMAND iot_host_demo)
endif()
