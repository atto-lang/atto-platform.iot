cmake_minimum_required(VERSION 3.20)
project(ores_iot_frame VERSION 0.1.0 LANGUAGES C)
include(CTest)

target_include_directories(ores_iot_frame PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(ores_iot_frame PUBLIC c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ores_iot_frame PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
